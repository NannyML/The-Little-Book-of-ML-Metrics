% First page of a metric is verso (left/even); the next page is recto.
% Padding is inserted only when necessary and carries neither folio nor band.
\newcommand{\metricentry}[2]{%
 \clearpage
 \ifodd\value{page}\thispagestyle{empty}\null\newpage\fi
 \metricentrytitle{#1}{#2}\global\metricstartpage=\value{page}}
\AddToHook{shipout/background}{%
 \ifmetricentrypage
 \begin{tikzpicture}[remember picture,overlay]
 \ifodd\value{page}
  \fill[\metricband] ([xshift=-13.425mm]current page.north east) rectangle (current page.south east);
  \node[rotate=90,anchor=east,inner sep=0pt,font=\sffamily\fontsize{8}{10}\selectfont]
   at ([xshift=-11.675mm,yshift=-13.175mm]current page.north east) {\chaptertitle\ \textendash\ \sectiontitle};
 \else
  \fill[\metricband] (current page.north west) rectangle ([xshift=13.425mm]current page.south west);
  \node[rotate=90,anchor=east,inner sep=0pt,font=\sffamily\fontsize{8}{10}\selectfont]
   at ([xshift=11.675mm,yshift=-13.175mm]current page.north west) {\chaptertitle\ \textendash\ \sectiontitle};
 \fi
 \end{tikzpicture}
 \global\metricentrypagefalse
 \fi}

% Fail early if later editing breaks a facing-page metric spread.
\newcount\metricstartpage
\newcommand{\metricsecondpage}{%
 \par\clearpage\typeout{SPREADFIRST|\thesection|\the\metricstartpage|\thepage}%
 \ifnum\value{page}=\numexpr\metricstartpage+1\relax\else
  \PackageError{hardcover}{First page of metric \sectiontitle\space overflowed}{Keep the explanation and comparison boxes on one page.}\fi}
\newcommand{\metricend}{%
 \par\clearpage\typeout{SPREADEND|\thesection|\the\metricstartpage|\thepage}%
 \ifnum\value{page}=\numexpr\metricstartpage+2\relax\else
  \PackageError{hardcover}{Second page of metric \sectiontitle\space overflowed}{Keep the visual and supporting discussion on one page.}\fi}

% Fraction and radical rules share the math extension font's rule thickness.
% Set every configured math size so fitted equations retain a printable rule.
\makeatletter
\AtBeginDocument{%
 \begingroup
 \@for\bookmathsize:=8,9,10,11,12,14.4\do{%
  \fontsize{\bookmathsize}{18}\selectfont\check@mathfonts
  \fontdimen8\textfont3=.9pt
  \fontdimen8\scriptfont3=.9pt
  \fontdimen8\scriptscriptfont3=.9pt}
 \endgroup}
\makeatother

% A chapter opener must keep its introduction and contents on one page.
\newcount\bookopenerstart
\let\hardcoverchapteropener\chapteropener
\renewcommand{\chapteropener}[5]{%
 \clearpage\global\bookopenerstart=\value{page}%
 \hardcoverchapteropener{#1}{#2}{#3}{#4}{#5}%
 \typeout{CHAPTERFIT|\thechapter|\the\bookopenerstart|\thepage}%
 \ifnum\value{page}=\numexpr\bookopenerstart+1\relax\else
  \PackageError{hardcover}{Chapter opener #1 overflowed}{Keep the introduction and contents on one page.}\fi}
