\newcommand{\bookleftmargin}{16mm}
\newcommand{\booksetleftmargin}{\ifodd\value{page}\renewcommand{\bookleftmargin}{16mm}\else\renewcommand{\bookleftmargin}{17.175mm}\fi}
\patchcmd{\chapteropener}{\clearpage}{\clearpage\booksetleftmargin}{}{}
% First page of a metric is verso (left/even); the next page is recto.
% Padding is inserted only when necessary and carries neither folio nor band.
\newcommand{\metricentry}[2]{%
 \clearpage
 \ifodd\value{page}\thispagestyle{empty}\null\newpage\fi
 \metricentrytitle{#1}{#2}\global\metricstartpage=\value{page}}
% Record the first title line's cap height without changing its layout.
\newlength{\metrictitlecapheight}
\renewcommand{\metrictitleanchor}{%
 \setlength{\metrictitlecapheight}{\fontcharht\font`M}%
 \tikz[remember picture,overlay,baseline=0pt]{%
  \coordinate (metric-title-\thepage) at (0,\the\metrictitlecapheight);}}
% The 10.75 mm trimmed band leaves a 3.25 mm white gutter to the
% 14 mm outer text margin. Keep the label 8.05 mm from trim, leaving about
% 1.3 mm of coloured padding on its text-facing side. Fixed height/depth
% prevent chapter names with different ascenders and descenders from
% changing its horizontal position.
\tikzset{metric edge label/.style={rotate=90,anchor=east,
 inner sep=0pt,outer sep=0pt,text height=6.08pt,text depth=1.92pt,
 font=\sffamily\fontsize{8}{10}\selectfont}}
% Edge navigation is visible only on left/even pages, so readers browse
% labels from one side of the closed book. Right-hand entries stay unbanded.
\AddToHook{shipout/background}{%
 \ifmetricentrypage
 \ifodd\value{page}\else
  \begin{tikzpicture}[remember picture,overlay]
   \fill[\metricband] (current page.north west) rectangle ([xshift=13.925mm]current page.south west);
   \node[metric edge label]
    at ([xshift=11.225mm]current page.north west |- metric-title-\thepage) {\chaptertitle\ \textendash\ \sectiontitle};
  \end{tikzpicture}
 \fi
 \global\metricentrypagefalse
 \fi}
% Position chapter furniture within the same mirrored safe area as the prose.

\patchcmd{\chapteropener}{yshift=-1.1cm}{yshift=-14.175mm}{}{}
\patchcmd{\chapteropener}{yshift=-4.9cm}{yshift=-52.175mm}{}{}
% Fine rules must survive print at their final inclusion scale.
\tikzset{semithick/.style={line width=.95pt}}
\forestset{dmap/.append style={for tree={edge={draw,line width=.8pt,black!40}}}}

% Fail early if later editing breaks a facing-page metric spread.
\newcount\metricstartpage
\newcommand{\metricsecondpage}{%
 \par\clearpage\typeout{SPREADFIRST|\thesection|\the\metricstartpage|\thepage}%
 \ifnum\value{page}=\numexpr\metricstartpage+1\relax\else
  \PackageError{hardcover}{First page of metric \sectiontitle\space overflowed}{Keep the explanation and comparison boxes on one page.}\fi}
\newcommand{\metricend}{%
 \par\clearpage\typeout{SPREADEND|\thesection|\the\metricstartpage|\thepage}%
 \ifnum\value{page}=\numexpr\metricstartpage+2\relax\else
  \PackageError{hardcover}{Second page of metric \sectiontitle\space overflowed}{Keep the visual and supporting discussion on one page.}\fi}

% mathematical scripts above seven PDF points, including formula fit scaling.

% Fraction and radical rules share the math extension font's rule thickness.
% Set every configured math size so fitted equations retain a printable rule.
\makeatletter
\AtBeginDocument{%
 \begingroup
 \@for\bookmathsize:=8,9,10,11,12,14.4\do{%
  \fontsize{\bookmathsize}{18}\selectfont\check@mathfonts
  \fontdimen8\textfont3=.9pt
  \fontdimen8\scriptfont3=.9pt
  \fontdimen8\scriptscriptfont3=.9pt}
 \endgroup}
\makeatother

% A chapter opener must keep its introduction and contents on one page.
\newcount\bookopenerstart
\let\hardcoverchapteropener\chapteropener
\renewcommand{\chapteropener}[5]{%
 \clearpage\global\bookopenerstart=\value{page}%
 \hardcoverchapteropener{#1}{#2}{#3}{#4}{#5}%
 \typeout{CHAPTERFIT|\thechapter|\the\bookopenerstart|\thepage}%
 \ifnum\value{page}=\numexpr\bookopenerstart+1\relax\else
  \PackageError{hardcover}{Chapter opener #1 overflowed}{Keep the introduction and contents on one page.}\fi}
