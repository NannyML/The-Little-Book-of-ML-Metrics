    dmap/.style={
        for tree={
            grow'=0, child anchor=west, parent anchor=east, anchor=west,
            font=\footnotesize\nohyph, text=black!80, node options={align=left},
            l sep=3.5mm, s sep=3mm, inner xsep=0pt, inner ysep=1pt, outer xsep=1.5pt,
            edge={draw, line width=0.5pt, black!30},
            forked edges, fork sep=2mm,
            tier/.option=level,
        },
        where n children=0{tier=leaf, text width=3.95cm}{},
        % with an odd number of children the parent sits level with the middle one, so its stub and that child's tick form one line
        where n children=3{calign=child, calign child=2}{},
        where n children=5{calign=child, calign child=3}{},
        where level=1{fork sep=0pt, if n children=0{}{text width=2.2cm}}{},
        where level=2{if n children=0{}{text width=2.3cm}}{},
        where level=3{if n children=0{}{text width=2.3cm}}{},
    },
    % one spacing between answer blocks in every map; compact only tightens the tree's levels

    wide/.style={

    },

    },

    \clearpage
    \mapstartpage=\value{page}%
    \colorlet{mapband}{#1}\colorlet{mapaccent}{#2}%

    \thispagestyle{mapstyle}%

    \vspace{10pt}
    \noindent

    \clearpage
    \typeout{MAPFIT|\the\mapstartpage|\thepage}%

    \fi
}
